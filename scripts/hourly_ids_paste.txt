sudo docker exec -i infra-api-1 python - <<'EOF'
# 날짜별·시간대별로 앱이 켜져 있던 회원 (heartbeat/app_health/app_start 기준, 스모크 CP-S… 제외). 읽기 전용.
# 열: 날짜 | 시간대(KST) | 대수 | 켜져 있던 회원번호 (첫 신호~마지막 신호) | 비고
import os, psycopg
from datetime import date, timedelta
DATES = ["2026-10-27", "2026-10-28"]           # 뽑을 날짜
TEAM  = {"CP-B83C408F": "민서"}                 # 팀원 표시용 (비고)
SINCE = "2026-10-27 16:00"                     # 삭제 조치 공지 시각 (비고)
c = psycopg.connect(os.environ["DATABASE_URL"])
rows = c.execute("""
SELECT to_char(e.received_at AT TIME ZONE 'Asia/Seoul', 'YYYY-MM-DD') d,
       to_char(date_trunc('hour', e.received_at AT TIME ZONE 'Asia/Seoul'), 'HH24') h,
       u.cp_number,
       to_char(MIN(e.received_at AT TIME ZONE 'Asia/Seoul'), 'HH24:MI') first_seen,
       to_char(MAX(e.received_at AT TIME ZONE 'Asia/Seoul'), 'HH24:MI') last_seen
FROM events e JOIN users u USING(user_id)
WHERE e.received_at >= %s::timestamptz
  AND e.received_at <  %s::timestamptz
  AND e.event IN ('heartbeat','app_health','app_start')
  AND u.cp_number NOT LIKE 'CP-S%%'
GROUP BY 1, 2, 3 ORDER BY 1, 2, 3
""", (DATES[0] + " 00:00+09", (date.fromisoformat(DATES[-1]) + timedelta(days=1)).isoformat() + " 00:00+09")).fetchall()
by = {}
for d, h, cp, a, b in rows:
    by.setdefault((d, h), []).append((cp, a, b))
print("| 날짜 | 시간대(KST) | 대수 | 켜져 있던 회원번호 (첫 신호~마지막 신호) | 비고 |")
print("|---|---|---|---|---|")
for d in DATES:
    for h in range(24):
        hh = f"{h:02d}"
        lst = by.get((d, hh), [])
        if not lst: continue
        ids = ", ".join(f"{cp}({a}~{b})" for cp, a, b in lst)
        note = []
        team = [TEAM[cp] for cp, _, _ in lst if cp in TEAM]
        if team: note.append("팀원 " + "·".join(team) + " 포함")
        ext = len(lst) - len(team)
        note.append(f"외부 {ext}대")
        if f"{d} {hh}:00" >= SINCE: note.append("삭제 조치 공지 후")
        print(f"| {d} | {hh}시 | {len(lst)} | {ids} | {' / '.join(note)} |")
EOF
